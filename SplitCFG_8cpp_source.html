29void SplitCFG::dfs_visit_wait(
30 const SplitCFGBlock* BB,
31 llvm::SmallPtrSet<const SplitCFGBlock*, 32>& visited_blocks,
32 llvm::SmallVectorImpl<const SplitCFGBlock*>& waits_to_visit,
33 llvm::SmallPtrSetImpl<const SplitCFGBlock*>& visited_waits,
34 llvm::SmallVector<std::pair<const SplitCFGBlock*, SplitCFGPathInfo>>&
35 curr_path) {
37 if (BB->succ_empty()) {
38 return;
39 }
40
41 popping_ = false;
42 // successors to visit
43 llvm::SmallVector<
44 std::pair<const SplitCFGBlock*, SplitCFGBlock::const_succ_iterator>, 8>
45 to_visit;
46
47 visited_blocks.insert(BB);
48 to_visit.push_back(std::make_pair(BB, BB->succ_begin()));
49
50 do {
51 std::pair<const SplitCFGBlock*, SplitCFGBlock::const_succ_iterator>& Top =
52 to_visit.back();
53 const SplitCFGBlock* ParentBB{Top.first};
54 SplitCFGBlock::const_succ_iterator& I{Top.second};
55
57 bool bb_has_wait{(ParentBB->hasWait())};
58 bool cond_block{isConditional(ParentBB)};
59
61 if (!popping_) {
62 llvm::dbgs() << "Visit Parent BB# " << ParentBB->getBlockID()
63 << " isCondition: " << cond_block << " ";
64 curr_path.push_back(std::make_pair(ParentBB, SplitCFGPathInfo{ParentBB}));
65 } else {
66 }
67
70 if (bb_has_wait) {
71 llvm::dbgs() << " has WAIT ";
72 bool found_suc{getUnvisitedSuccessor(ParentBB, I, visited_blocks, BB)};
73 // A wait to be visited is added.
74 if (visited_waits.insert(ParentBB).second) {
75 waits_to_visit.insert(waits_to_visit.begin(), BB);
76 ++next_state_count_;
77 auto wit{wait_next_state_.find(ParentBB)};
78 if (wit == wait_next_state_.end()) {
79 wait_next_state_.insert(
80 std::make_pair(ParentBB, std::make_pair(BB, next_state_count_)));
81 }
82 }
83 }
84
85 // If there is a successor that has not been visited, then remember that
86 // block.
87 dumpVisitedBlocks(visited_blocks);
88 bool found_succ{getUnvisitedSuccessor(ParentBB, I, visited_blocks, BB)};
89 llvm::dbgs() << "Found successor BB " << BB->getBlockID()
90 << " for parentBB " << ParentBB->getBlockID() << "\n";
91
95 if (cond_block && !bb_has_wait) {
99 if (const auto true_block_succ = *ParentBB->succ_begin()) {
100 auto path_info = std::find_if(
101 curr_path.begin(), curr_path.end(),
102 [ParentBB](
103 const std::pair<const SplitCFGBlock*, SplitCFGPathInfo>& x) {
104 return x.first->getBlockID() == ParentBB->getBlockID();
105 });
106
107 if (true_block_succ == BB) {
108 true_path_ = true;
109 llvm::dbgs() << "Going down the TRUE path for BB"
110 << path_info->first->getBlockID() << " \n";
111
112 } else {
113 llvm::dbgs() << "Going down the FALSE path for BB"
114 << path_info->first->getBlockID() << " \n";
115
116 if (true_path_) {
118 }
119 }
120 }
121 }
122
123 // Use the recursive call for loops with 2 successors, and IF blocks.
124 if ((isLoopWithTwoSuccessors(ParentBB) || isConditional(ParentBB) ) && found_succ) {
125 do {
126 // llvm::dbgs() << "\n==============================================";
127 llvm::dbgs() << "\n#### BB " << ParentBB->getBlockID()
128 << " is a loop with 2 succ\n";
129 llvm::SmallPtrSet<const SplitCFGBlock*, 32> loop_visited_blocks{
130 visited_blocks};
131 // ParentBB has been visited so don't revisit it
132 loop_visited_blocks.insert(ParentBB);
133 dumpVisitedBlocks(loop_visited_blocks);
134 visited_blocks.insert(BB);
135 llvm::dbgs() << "\n==============================================";
136 llvm::dbgs() << "\nRecurse DFS starting at BB " << BB->getBlockID()
137 << " visited_block size " << visited_blocks.size() << "\n";
138 dfs_visit_wait(BB, loop_visited_blocks, waits_to_visit, visited_waits,
139 curr_path);
140 llvm::dbgs() << "\n";
141
144 // updateVisitedBlocks(loop_visited_blocks, loop_visited_blocks);
145 // dumpVisitedBlocks(visited_blocks);
146 llvm::dbgs() << "\nEND Recurse DFS"
147 << " visited_block size " << visited_blocks.size() << "\n";
148 llvm::dbgs() << "\n==============================================";
149
158 //
159
160 found_succ = getUnvisitedSuccessor(ParentBB, I, visited_blocks, BB);
161 // llvm::dbgs() << "unvisited BB " << BB->getBlockID() << " is succ "
162 // << found_succ << "\n";
163 //
166 if (!found_succ) {
167 updateVisitedBlocks(visited_blocks, loop_visited_blocks);
168 }
169 } while (found_succ);
170
171 } else {
172 // Recursive call in the if-then will traverse the subgraph.
173 // Only insert successor if recursive call does not visit subgraph.
174 }
175 addSuccessorToVisitOrPop(bb_has_wait, BB, to_visit, found_succ);
176 // std::cin.get();
177 // llvm::dbgs() << "to_visit ";
178 // dumpSmallVector(to_visit);
179 // llvm::dbgs() << " End loop \n";
180 } while (!to_visit.empty());
181}

400void SplitCFG::splitBlock(clang::CFGBlock* block) {
401 assert(block != nullptr);
402
403 unsigned int num_elements{block->size()};
404
405 llvm::dbgs() << "\n\nChecking if block " << block->getBlockID()
406 << " has a wait()\n";
407 llvm::dbgs() << "Number of elements " << num_elements << "\n";
408 // We are going to have two vectors.
409 // 1. A vector of vector pointers to CFGElements.
410 // 2. A vector of pointers to CFGElements that are waits.
411 //
412
413 VectorCFGElementPtr vec_elements{};
414 llvm::SmallVector<std::pair<VectorCFGElementPtr, bool>> split_elements;
415
417 if (num_elements == 0) {
418 split_elements.push_back(std::make_pair(vec_elements, false));
419 } else {
420 bool has_wait{false};
421 for (auto const& element : block->refs()) {
422 has_wait = false;
423 llvm::dbgs() << "Is this a wait? ";
424 element->dump();
425
429 const clang::CFGElement* element_ptr{element.operator->()};
431 if (isElementWait(*element)) {
432 llvm::dbgs() << " IT IS A WAIT\n";
434 if (num_elements == 1) {
435 llvm::dbgs() << "DBG: Only one statement and it is a wait().\n";
436 }
437
438 // Add all elements before the wait into the split elements.
439 if (vec_elements.size() != 0) {
440 split_elements.push_back(std::make_pair(vec_elements, false));
441 vec_elements.clear();
442 }
443
445 vec_elements.push_back(element_ptr);
446 split_elements.push_back(std::make_pair(vec_elements, true));
447 vec_elements.clear();
448
449 has_wait = true;
450 } else {
451 vec_elements.push_back(element_ptr);
452 }
453 }
454
455 if (vec_elements.size() != 0) {
456 split_elements.push_back(std::make_pair(vec_elements, has_wait));
457 }
458 }
459
460 createWaitSplitCFGBlocks(block, split_elements);
461}

463void SplitCFG::createWaitSplitCFGBlocks(
464 clang::CFGBlock* block,
465 const llvm::SmallVectorImpl<std::pair<SplitCFG::VectorCFGElementPtr, bool>>&
466 split_elements) {
467 // dumpSplitElements(split_elements);
469 llvm::dbgs() << "Number of entries in split_elements "
470 << split_elements.size() << "\n";
471 unsigned int id{0};
472 SplitCFGBlock* prev_block{nullptr};
473
474 for (auto const& elements : split_elements) {
475 llvm::dbgs() << "Element number " << id << " ";
477 SplitCFGBlock* new_split{nullptr};
478
479 bool already_exists{false};
480 // id = 0 is when it is the first sequence of elements.
481 if (id == 0) {
482 auto scit{sccfg_.find(block->getBlockID())};
483 // Must be true.
484 // bool already_exists{scit != sccfg_.end()};
485 new_split = scit->second;
486 new_split->id_ = block->getBlockID();
487
488 // Successor: succ block from CFGBlock (overridden)
490 if (split_elements.size() == 1) {
491 addSuccessors(new_split, block);
492 }
493 // Prececessor: prev block from CFGBlock
494 addPredecessors(new_split, block);
495 } else {
496 new_split = new SplitCFGBlock{};
497 new_split->id_ = block->getBlockID() * 10 + id;
498
500 //
502 llvm::dbgs() << "SB# " << new_split->id_ << " size "
503 << split_elements.size() << "\n";
504 if (id == split_elements.size() - 1) {
505 llvm::dbgs() << " add successors of BB# " << block->getBlockID()
506 << "\n";
507 addSuccessors(new_split, block);
508 }
509 prev_block->successors_.push_back(new_split);
511 new_split->predecessors_.push_back(prev_block);
512 sccfg_.insert(std::make_pair(block->getBlockID() * 10 + id, new_split));
513 }
514
515 new_split->block_ = block;
516 new_split->has_wait_ = elements.second;
517 new_split->elements_ = elements.first;
518
519 // If it's a wait() then get the argument.
520 if (new_split->has_wait_) {
522 const clang::CFGElement* wait_el = (*new_split->elements_.begin());
523 new_split->wait_arg_ = getWaitArgument(*wait_el);
524 }
525
528 //
529 if (isLoop(new_split)) {
530 new_split->identifyBreaks(context_);
531 }
532 // Set the successor of new_split.
533 prev_block = new_split;
534 ++id;
535 }
536}

635void SplitCFG::createUnsplitBlocks() {
636 for (auto begin_it = cfg_->nodes_begin(); begin_it != cfg_->nodes_end();
637 ++begin_it) {
638 auto block{*begin_it};
639 SplitCFGBlock* new_block{new SplitCFGBlock{}};
640 new_block->id_ = block->getBlockID();
641
643 auto stmt{block->getTerminatorStmt()};
644 new_block->is_conditional_ = stmt && llvm::isa<clang::IfStmt>(stmt);
645
646 bool last_succ_is_null{false};
647 if (block->succ_size() == 2) {
648 // Check that the last one is not NULL
649 if (*block->succ_rbegin() == nullptr) {
650 last_succ_is_null = true;
651 }
652 }
653
655 bool is_loop{stmt && (llvm::isa<clang::WhileStmt>(stmt) ||
656 llvm::isa<clang::ForStmt>(stmt) ||
657 llvm::isa<clang::DoStmt>(stmt))};
658
659 new_block->is_loop_with_two_succ_ =
660 (stmt && is_loop && (last_succ_is_null == false));
661
662 sccfg_.insert(std::make_pair(new_block->id_, new_block));
663 }
664
665 llvm::dbgs() << "sccfg( " << sccfg_.size() << ") ids: ";
666 for (auto const& entry : sccfg_) {
667 llvm::dbgs() << entry.first << " ";
668 }
669 llvm::dbgs() << "\n";
670}
